    <!--
      GeLaCo bi-objective configuration for ECF 1.6.1.

      Paper (arXiv:2507.10059 §4) target:
        Algorithm        : NSGA-II
        Population       : 200 individuals
        Termination      : 30,000 fitness evaluations
        Crossover        : integer-adapted SBX, p=0.9, eta_c=20.0
        Mutation         : polynomial, p=1/(3L)=1/96, eta_m=20
        Encoding         : 3L = 96 vars; b,e in [0, L-1], a in {0,1}
                           (the {0,1} projection is handled in the C++ decoder)
        Calibration      : 64 Wikipedia sentences (handled by server.py)

      Deviations forced by the operator set in ECF 1.6.1:
        * SBX probability set to 1.0 (no other crossover registered; the
          framework requires per-genotype crossover probs sum to 1.0).
        * SBX distribution index registered as `crx.sbx.ni` (UINT) in the
          global Registry, not as a genotype entry, and defaults to 1; we
          set it to 20 to match the paper.
        * Polynomial mutation is not implemented in this ECF; the only
          FloatingPoint mutation operator is `mut.simple`, which uniformly
          replaces ONE random gene per call. We set per-individual mutation
          probability (`mutation.indprob`) to 1.0 so every individual gets
          one gene replaced — same expected count of mutated genes as the
          paper's polynomial mutation at p=1/96 per gene (96 * 1/96 = 1),
          but with uniform replacement instead of small local perturbation.

      The Python evaluator (server.py) is started by the PBS job script
      and connected over two named FIFOs (env vars GELACO_REQ_FIFO,
      GELACO_RSP_FIFO, GELACO_READY_FILE).
    -->

    <Algorithm>
        <!-- NSGA-II takes no registered parameters in ECF 1.6.1. -->
        <NSGA2/>
    </Algorithm>

    <Genotype>
        <FloatingPoint>
            <Entry key="dimension">96</Entry>
            <Entry key="lbound">0.0</Entry>
            <Entry key="ubound">31.0</Entry>

            <!-- Crossover: only SBX registered. Must sum to 1.0 across all
                 crossover ops for this genotype. -->
            <Entry key="crx.sbx">1.0</Entry>

            <!-- Mutation: only mut.simple registered. Weight is normalized
                 across mutation ops for this genotype; setting to 1.0

            <Entry key="mut.simple">1.0</Entry>
        </FloatingPoint>
    </Genotype>

    <Registry>
